        {% set version_num = ver.version_number if ver.version_number is defined else ver['version_number'] if 'version_number' in ver else ver.version if ver.version is defined else ver['version'] if 'version' in ver else (versions|length - i - 1) %}
        <option value="/regulation/{{ regulation.id }}/view/version/{{ version_num }}" 
                {% if version_num == current_version_num %}selected{% endif %}>
            Version {{ version_num }}{% if i == 0 %} (最新){% endif %}
        </option>
        {% endfor %}
    </select>
</div>
{% endif %}
